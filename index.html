  /* 名牌自適應排版：量測名單區實際大小，在 2–12 欄中挑字最大的排法；

     字級差距約 15% 以內時，優先選空格最少、名牌比例最協調者。視窗大小改變時自動重排。 */

  function fit(n, hasD) {

    const box = $('#body').getBoundingClientRect();

    const W = Math.max(200, box.width - 2), H = Math.max(120, box.height - 2);

    const u = Math.min(innerWidth, innerHeight * 1.778) / 100;

    const cands = [];

    for (let c = 2; c <= 12; c++) {

      const r = Math.ceil(n / c), w = W / c, h = H / r;

      // 名字最多約 4.5 字寬；有選區時多一行小字，名字可用高度較少

      const nf = Math.min(2.8 * u, w / 5, h / (hasD ? 2.2 : 1.6));

      cands.push({ c, nf, w, waste: c * r - n, shape: Math.abs(Math.log((w / h) / 2.2)) });

    }

    // 字級在最大值 85% 以內者，優先選空格最少、名牌比例（長寬約 2.2）最協調者

    const maxNf = Math.max(...cands.map(x => x.nf));

    const best = cands.filter(x => x.nf >= maxNf * 0.85)

      .sort((a, b) => a.waste - b.waste || a.shape - b.shape || b.nf - a.nf)[0];

    const df = Math.min(1.1 * u, Math.max(0.7 * u, best.nf * 0.42), best.w / 7.5);

    return { cols: best.c, nf: Math.floor(best.nf), df: Math.floor(df) };

  }
